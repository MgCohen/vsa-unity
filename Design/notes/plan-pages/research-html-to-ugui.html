HTML to uGUI

A browser snapshot to JSON, then an Editor importer, gives a deterministic uGUI prefab.

VIABLE browser snapshot + importer
RISK Figma converters, AI and MCP tools
SPIKE runtime HTML UI: one UI, Unity shows it (ReactUnity, Mako UI)
DEAD END HTML parsers, desktop-only browsers

Two routes fit. Route A: the HTML becomes uGUI prefabs (flow below). Route B: the HTML is the UI, and a runtime engine shows it (Render HTML tab).

For route A, only a real browser computes CSS layout correctly, so let the browser do the layout. A small open project (HTML-to-UGUI-Baker) uses this method already. vsa-unity can own a short JS extractor and a C# importer with no paid tool.

// Runs in the page after fonts load. Writes JSON into the DOM.
const rgba = c => (c.match(/[\d.]+/g) || [0,0,0,0]).map(Number);
const nodes = [];
function walk(el, parent) {
  const s = getComputedStyle(el), r = el.getBoundingClientRect();
  if (s.display === "none") return;
  const i = nodes.push({
    name: el.dataset.u || el.tagName.toLowerCase(), parent,
    x: r.left, y: r.top, w: r.width, h: r.height,
    bg: rgba(s.backgroundColor), radius: parseFloat(s.borderTopLeftRadius),
    text: el.children.length ? "" : el.textContent.trim(),
    font: s.fontFamily, size: parseFloat(s.fontSize),
    color: rgba(s.color), align: s.textAlign,
    flex: s.display.includes("flex") ? s.flexDirection : ""
  }) - 1;
  for (const c of el.children) walk(c, i);
}
document.fonts.ready.then(() => {
  walk(document.querySelector("[data-u-root]"), -1);
  const out = document.createElement("pre");
  out.id = "layout"; out.textContent = JSON.stringify({ nodes });
  document.body.append(out);
});

Step 4 · LayoutImporter.cs 
// Flat list keeps JsonUtility happy (no recursive types).
[System.Serializable] class Node { public string name, text, font, align, flex;
  public int parent; public float x, y, w, h, radius, size;
  public float[] bg, color; }
[System.Serializable] class Layout { public Node[] nodes; }

static GameObject Import(string json) {
  var nodes = JsonUtility.FromJson<Layout>(json).nodes;
  var rts = new RectTransform[nodes.Length];
  for (int i = 0; i < nodes.Length; i++) {
    var n = nodes[i];
    var p = n.parent < 0 ? null : nodes[n.parent];
    var rt = new GameObject(n.name, typeof(RectTransform)).GetComponent<RectTransform>();
    if (p != null) rt.SetParent(rts[n.parent], false);
    rt.anchorMin = rt.anchorMax = rt.pivot = new Vector2(0, 1);
    rt.anchoredPosition = new Vector2(n.x - (p?.x ?? 0), -(n.y - (p?.y ?? 0)));
    rt.sizeDelta = new Vector2(n.w, n.h);
    if (n.bg.Length == 4 && n.bg[3] > 0)
      rt.gameObject.AddComponent<Image>().color = Rgba(n.bg);
    if (n.text != "") AddText(rt, n); // child object: one Graphic per GameObject
    rts[i] = rt;
  }
  return rts[0].gameObject;
}

// Sketch. The ReactUnity member names are not confirmed.
public sealed class CounterRuntimeView : PresenterComponent<CounterPresenter>
{
    [SerializeField] ReactRendererUGUI react;

    protected override void OnBind()
    {
        base.OnBind();
        // UI → C#: the JSX calls Globals.roll()
        react.Globals["roll"] = (Action)(() => _ = Presenter.RollAsync());
    }

    // C# → UI: runs on each Changed
    protected override void Render()
    {
        react.Globals["count"] = Presenter.Count;
        react.Globals["last"]  = Presenter.Last;
    }
}
